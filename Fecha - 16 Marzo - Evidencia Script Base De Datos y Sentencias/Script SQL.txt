DROP DATABASE IF EXISTS deliweb;

CREATE DATABASE deliweb
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE deliweb;

-- ============================================================
-- TABLA: ROLES
-- ============================================================

CREATE TABLE roles (
    id_rol INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE
);

-- ============================================================
-- TABLA: USUARIOS
-- ============================================================

CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    correo VARCHAR(150) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    telefono VARCHAR(20),
    direccion VARCHAR(200),
    id_rol INT NOT NULL,
    estado BOOLEAN DEFAULT TRUE,

    FOREIGN KEY (id_rol)
        REFERENCES roles(id_rol)
);

-- ============================================================
-- TABLA: CATEGORIAS
-- ============================================================

CREATE TABLE categorias (
    id_categoria INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion VARCHAR(255),
    estado BOOLEAN DEFAULT TRUE
);

-- ============================================================
-- TABLA: PRODUCTOS
-- ============================================================

CREATE TABLE productos (
    id_producto INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    descripcion TEXT,
    precio DECIMAL(10,2) NOT NULL,
    stock INT NOT NULL DEFAULT 0,
    imagen VARCHAR(255),
    id_categoria INT NOT NULL,
    estado BOOLEAN DEFAULT TRUE,

    FOREIGN KEY (id_categoria)
        REFERENCES categorias(id_categoria)
);

-- ============================================================
-- SENTENCIAS DML
-- ============================================================

-- CONSULTAR TODOS LOS PRODUCTOS

SELECT * FROM productos;

-- CONSULTAR PRODUCTOS DISPONIBLES

SELECT
    id_producto,
    nombre,
    precio,
    stock
FROM productos
WHERE estado = TRUE;

-- BUSCAR PRODUCTOS DE MÁS DE $10.000

SELECT
    nombre,
    precio
FROM productos
WHERE precio > 10000;

-- ACTUALIZAR EL PRECIO DE UN PRODUCTO

UPDATE productos
SET precio = 47000
WHERE id_producto = 1;

-- ACTUALIZAR EL STOCK

UPDATE productos
SET stock = stock + 10
WHERE id_producto = 1;

-- ELIMINAR UN PRODUCTO

DELETE FROM productos
WHERE id_producto = 5;


-- ============================================================
-- CONSULTA DE PRODUCTOS CON SU CATEGORIA
-- ============================================================

SELECT
    p.id_producto,
    p.nombre AS producto,
    c.nombre AS categoria,
    p.precio,
    p.stock
FROM productos p
INNER JOIN categorias c
    ON p.id_categoria = c.id_categoria;

-- ============================================================
-- PRODUCTOS DE LA CATEGORIA TORTAS
-- ============================================================

SELECT
    p.nombre,
    p.precio,
    c.nombre AS categoria
FROM productos p
INNER JOIN categorias c
    ON p.id_categoria = c.id_categoria
WHERE c.nombre = 'Tortas';

-- ============================================================
-- PRODUCTOS ORDENADOS DEL MÁS BARATO AL MÁS CARO
-- ============================================================

SELECT
    nombre,
    precio
FROM productos
ORDER BY precio ASC;

-- ============================================================
-- PRODUCTOS ORDENADOS DEL MÁS CARO AL MÁS BARATO
-- ============================================================

SELECT
    nombre,
    precio
FROM productos
ORDER BY precio DESC;
